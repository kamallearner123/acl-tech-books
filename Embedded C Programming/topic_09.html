Topic 9

Make and CMake Basics

Make expresses file dependencies and rebuilds only what changed. CMake describes targets and generates a native build system.

CC=gcc
CFLAGS=-std=c11 -Wall -Wextra
app: main.o gpio.o
	$(CC) $(CFLAGS) $^ -o $@
%.o: %.c
	$(CC) $(CFLAGS) -c $< -o $@
clean:
	rm -f *.o app
cmake_minimum_required(VERSION 3.16)
project(app C)
add_executable(app main.c gpio.c)
target_compile_options(app PRIVATE -Wall -Wextra)
