Module 5 Capstone Project

Layered Architecture & Production Integration

The hallmark of an elite embedded software engineer is the ability to design hardware-independent architectures. When business requirements force an MCU migration from ST to NXP or Texas Instruments, a poorly architected codebase must be scrapped and rewritten. A professionally layered architecture isolates the application and driver business logic behind clean abstraction interfaces, allowing 100% of driver and application code to be tested on Linux/WSL and seamlessly deployed to STM32 hardware with only a swap of the bottom-most HAL adapter layer.

Firmware Layer Responsibility Portability Host (Linux/WSL) vs Target (STM32)
Application Telemetry formatting, threshold alerts, state coordination 100% Portable (Pure C) Identical C code on both platforms
Driver Layer BME280 sensor register protocol, calibration math 100% Portable (Pure C) Identical C code on both platforms
HAL Interface Function pointer contract for I2C/SPI/UART transfers Interface Definition Shared hal_i2c.h header
Platform Adapter Concrete hardware register or simulated bus transfer Hardware Specific Linux: hal_i2c_mock.c | STM32: hal_i2c_stm32.c

Part 1: Linux / WSL Project — Decoupled Sensor Subsystem & GitHub Actions CI/CD

We build a complete Environmental Sensor Subsystem. The driver consumes an abstract hal_i2c_t interface. On Linux/WSL, we inject a mock I2C driver that simulates sensor hardware registers in memory, enabling automated Unity unit tests and a production GitHub Actions CI workflow.

Step 1: Abstract Hardware Interface (hal_i2c.h)

/**
 * @file hal_i2c.h
 * @brief Platform-Agnostic I2C Hardware Abstraction Layer
 */
#ifndef HAL_I2C_H
#define HAL_I2C_H

#include <stdint.h>
#include <stdbool.h>
#include <stddef.h>

typedef enum {
    HAL_I2C_OK = 0,
    HAL_I2C_ERR_NACK,
    HAL_I2C_ERR_TIMEOUT,
    HAL_I2C_ERR_BUS_ERROR,
    HAL_I2C_ERR_PARAM
} hal_i2c_status_t;

/* Forward declaration of concrete instance */
typedef struct hal_i2c_bus hal_i2c_bus_t;

/* I2C Operations VTable */
typedef struct {
    hal_i2c_status_t (*write)(hal_i2c_bus_t *p_bus, uint8_t dev_addr, const uint8_t *data, size_t len);
    hal_i2c_status_t (*read)(hal_i2c_bus_t *p_bus, uint8_t dev_addr, uint8_t *data, size_t len);
    hal_i2c_status_t (*mem_read)(hal_i2c_bus_t *p_bus, uint8_t dev_addr, uint8_t reg_addr, uint8_t *data, size_t len);
} hal_i2c_ops_t;

struct hal_i2c_bus {
    const hal_i2c_ops_t *ops;
    void                *user_data; /* Points to hardware register base or mock buffer */
};

#endif /* HAL_I2C_H */

Step 2: Portable Sensor Driver (bme280.h & bme280.c)

This driver contains zero microcontroller-specific headers. It uses opaque pointers and handles all sensor calibration calculations: 

/**
 * @file bme280.c
 * @brief Fully Portable Sensor Driver using Abstract I2C HAL
 */
#include "bme280.h"

#define BME280_CHIP_ID_REG   0xD0U
#define BME280_EXPECTED_ID   0x60U
#define BME280_DATA_REG      0xF7U

typedef struct {
    hal_i2c_bus_t *i2c_bus;
    uint8_t        dev_addr;
    int32_t        calib_t1;
    int32_t        calib_t2;
} bme280_dev_t;

static bme280_dev_t g_sensors[2];
static uint8_t g_num_sensors = 0;

bme280_handle_t bme280_create(hal_i2c_bus_t *p_bus, uint8_t dev_addr) {
    if ((p_bus == NULL) || (g_num_sensors >= 2U)) {
        return NULL;
    }
    bme280_dev_t *dev = &g_sensors[g_num_sensors++];
    dev->i2c_bus = p_bus;
    dev->dev_addr = dev_addr;
    return (bme280_handle_t)dev;
}

bme280_status_t bme280_init(bme280_handle_t handle) {
    bme280_dev_t *dev = (bme280_dev_t *)handle;
    if ((dev == NULL) || (dev->i2c_bus->ops->mem_read == NULL)) {
        return BME280_ERR_PARAM;
    }

    uint8_t chip_id = 0U;
    hal_i2c_status_t status = dev->i2c_bus->ops->mem_read(dev->i2c_bus, dev->dev_addr, 
                                                           BME280_CHIP_ID_REG, &chip_id, 1U);
    if (status != HAL_I2C_OK) {
        return BME280_ERR_BUS;
    }

    if (chip_id != BME280_EXPECTED_ID) {
        return BME280_ERR_ID_MISMATCH;
    }

    return BME280_OK;
}

bme280_status_t bme280_read_temp_celsius(bme280_handle_t handle, int32_t *p_temp_c) {
    bme280_dev_t *dev = (bme280_dev_t *)handle;
    if ((dev == NULL) || (p_temp_c == NULL)) {
        return BME280_ERR_PARAM;
    }

    uint8_t raw_data[3];
    hal_i2c_status_t status = dev->i2c_bus->ops->mem_read(dev->i2c_bus, dev->dev_addr,
                                                           BME280_DATA_REG, raw_data, 3U);
    if (status != HAL_I2C_OK) {
        return BME280_ERR_BUS;
    }

    /* Convert 20-bit raw sensor data to milli-celsius */
    int32_t raw_adc = ((int32_t)raw_data[0] << 12) | ((int32_t)raw_data[1] << 4) | ((int32_t)raw_data[2] >> 4);
    *p_temp_c = (raw_adc * 100) / 1024; /* Simulated calibration curve */

    return BME280_OK;
}

Step 3: Host Mock I2C Bus & Unity Unit Test (test_sensor_subsystem.c)

/**
 * @file test_sensor_subsystem.c
 * @brief Host Unit Tests with Mock I2C Bus Implementation
 */
#include <stdio.h>
#include <assert.h>
#include "hal_i2c.h"
#include "bme280.h"

/* Mock Hardware Memory Space */
static uint8_t g_mock_registers[256];
static bool    g_inject_bus_timeout = false;

static hal_i2c_status_t mock_mem_read(hal_i2c_bus_t *bus, uint8_t dev, uint8_t reg, uint8_t *data, size_t len) {
    (void)bus; (void)dev;
    if (g_inject_bus_timeout) {
        return HAL_I2C_ERR_TIMEOUT;
    }
    for (size_t i = 0; i < len; i++) {
        data[i] = g_mock_registers[reg + i];
    }
    return HAL_I2C_OK;
}

static const hal_i2c_ops_t g_mock_ops = {
    .write = NULL,
    .read = NULL,
    .mem_read = mock_mem_read
};

int main(void) {
    printf("=== RUNNING HOST SENSOR SUBSYSTEM TESTS ===\n");

    hal_i2c_bus_t mock_bus = { .ops = &g_mock_ops, .user_data = NULL };
    bme280_handle_t sensor = bme280_create(&mock_bus, 0x76);
    assert(sensor != NULL);

    /* Test 1: Chip ID mismatch failure */
    g_mock_registers[0xD0] = 0x55; /* Bad ID */
    assert(bme280_init(sensor) == BME280_ERR_ID_MISMATCH);
    printf("  [PASS] Rejects invalid hardware Chip ID\n");

    /* Test 2: Valid Chip ID Initialization */
    g_mock_registers[0xD0] = 0x60; /* Valid BME280 Chip ID */
    assert(bme280_init(sensor) == BME280_OK);
    printf("  [PASS] Initializes correctly with valid Chip ID\n");

    /* Test 3: Hardware Bus Timeout Recovery */
    g_inject_bus_timeout = true;
    int32_t temp = 0;
    assert(bme280_read_temp_celsius(sensor, &temp) == BME280_ERR_BUS);
    printf("  [PASS] Correctly handles hardware bus timeout\n");

    printf("ALL HOST ARCHITECTURE TESTS PASSED!\n");
    return 0;
}

Step 4: Automated GitHub Actions CI Pipeline (.github/workflows/firmware_ci.yml)

name: Embedded Firmware CI Pipeline

on: [push, pull_request]

jobs:
  build-and-test:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout Code
        uses: actions/checkout@v3

      - name: Install Toolchain & Static Analyzers
        run: |
          sudo apt-get update
          sudo apt-get install -y gcc-arm-none-eabi cppcheck valgrind

      - name: Run MISRA-C Static Analysis
        run: |
          cppcheck --enable=warning,style --error-exitcode=1 src/

      - name: Run Host Unit Tests with AddressSanitizer
        run: |
          gcc -std=c11 -Wall -Wextra -Werror -fsanitize=address,undefined \
              src/bme280.c tests/test_sensor_subsystem.c -o test_runner
          ./test_runner

      - name: Cross-Compile Target Firmware for STM32
        run: |
          make target

      - name: Archive Binary Artifacts
        uses: actions/upload-artifact@v3
        with:
          name: stm32-firmware-binaries
          path: build/*.bin

Part 2: Physical STM32 Board Project — Production Flight Recorder & Data Logger

Now, we take the exact same bme280.c and bme280.h files and deploy them to an STM32F401RE microcontroller. We provide a concrete hardware implementation of hal_i2c_t using the STM32 hardware I2C1 peripheral (PB8/PB9).

Hardware Layer Implementation (hal_i2c_stm32.c)

/**
 * @file hal_i2c_stm32.c
 * @brief Concrete STM32 Register-Level I2C Hardware Driver
 */
#include "hal_i2c.h"

/* STM32F4 I2C1 Register Offsets */
#define RCC_APB1ENR   (*(volatile uint32_t *)(0x40023840UL))
#define RCC_AHB1ENR   (*(volatile uint32_t *)(0x40023830UL))
#define GPIOB_MODER   (*(volatile uint32_t *)(0x40020400UL))
#define GPIOB_AFRH    (*(volatile uint32_t *)(0x40020424UL))
#define GPIOB_OTYPER  (*(volatile uint32_t *)(0x40020404UL))

#define I2C1_CR1      (*(volatile uint32_t *)(0x40005400UL))
#define I2C1_CR2      (*(volatile uint32_t *)(0x40005404UL))
#define I2C1_OAR1     (*(volatile uint32_t *)(0x40005408UL))
#define I2C1_DR       (*(volatile uint32_t *)(0x40005410UL))
#define I2C1_SR1      (*(volatile uint32_t *)(0x40005414UL))
#define I2C1_SR2      (*(volatile uint32_t *)(0x40005418UL))
#define I2C1_CCR      (*(volatile uint32_t *)(0x4000541CUL))
#define I2C1_TRISE    (*(volatile uint32_t *)(0x40005420UL))

static hal_i2c_status_t stm32_i2c_mem_read(hal_i2c_bus_t *bus, uint8_t dev, uint8_t reg, uint8_t *data, size_t len) {
    (void)bus;
    /* 1. Generate Start Condition */
    I2C1_CR1 |= (1UL << 8); /* START */
    while (!(I2C1_SR1 & (1UL << 0))) { /* Wait for SB flag */ }

    /* 2. Send Device Write Address */
    I2C1_DR = (uint32_t)(dev << 1);
    while (!(I2C1_SR1 & (1UL << 1))) { /* Wait for ADDR flag */ }
    (void)I2C1_SR2; /* Clear ADDR flag by reading SR2 */

    /* 3. Send Register Address */
    I2C1_DR = (uint32_t)reg;
    while (!(I2C1_SR1 & (1UL << 7))) { /* Wait for TXE */ }

    /* 4. Generate Repeated Start for Read */
    I2C1_CR1 |= (1UL << 8); /* RESTART */
    while (!(I2C1_SR1 & (1UL << 0))) { /* Wait for SB */ }

    /* 5. Send Device Read Address */
    I2C1_DR = (uint32_t)((dev << 1) | 1U);
    while (!(I2C1_SR1 & (1UL << 1))) { /* Wait for ADDR */ }
    (void)I2C1_SR2;

    /* 6. Read Bytes */
    for (size_t i = 0; i < len; i++) {
        if (i == (len - 1U)) {
            I2C1_CR1 &= ~(1UL << 10); /* NACK last byte */
            I2C1_CR1 |= (1UL << 9);   /* STOP condition */
        } else {
            I2C1_CR1 |= (1UL << 10);  /* ACK */
        }
        while (!(I2C1_SR1 & (1UL << 6))) { /* Wait for RXNE */ }
        data[i] = (uint8_t)I2C1_DR;
    }

    return HAL_I2C_OK;
}

static const hal_i2c_ops_t g_stm32_i2c_ops = {
    .write = NULL,
    .read = NULL,
    .mem_read = stm32_i2c_mem_read
};

/* Exported Concrete I2C Bus Instance */
hal_i2c_bus_t g_stm32_i2c1_bus = {
    .ops = &g_stm32_i2c_ops,
    .user_data = (void *)0x40005400UL
};

Production Application Superloop (main_app.c)

/**
 * @file main_app.c
 * @brief Production Layered Environmental Flight Recorder
 */
#include <stdint.h>
#include "hal_i2c.h"
#include "bme280.h"

extern hal_i2c_bus_t g_stm32_i2c1_bus;

int main(void) {
    /* Initialize Hardware Peripherals (Clocks, GPIOs, I2C1) */
    
    /* Create portable sensor instance using the concrete STM32 HAL bus */
    bme280_handle_t sensor = bme280_create(&g_stm32_i2c1_bus, 0x76);
    
    if (bme280_init(sensor) != BME280_OK) {
        /* Fail-safe panic */
        while (1) { /* Diagnostic LED Flash */ }
    }

    while (1) {
        int32_t current_temp_mC = 0;
        bme280_status_t status = bme280_read_temp_celsius(sensor, &current_temp_mC);

        if (status == BME280_OK) {
            /* Package telemetry and log to non-volatile memory or stream over UART */
        }

        /* Power Management: Enter Sleep Mode until next timer tick */
        __asm volatile ("wfi");
    }

    return 0;
}

Hands-on Engineering Challenges

Challenge 1 (Host)

SPI Bus Mock & Flash Memory Driver

Design a hal_spi.h interface. Implement a driver for an external SPI Flash chip (e.g. W25Q128) and create a host-based mock with a 1MB RAM buffer verifying page write and sector erase operations.

Challenge 2 (Host)

CMock Automated Mock Generation

Integrate CMock into your Linux build. Automatically generate mock header and source files directly from hal_i2c.h and verify argument expectations with Unity.

Challenge 3 (STM32)

Multi-Device I2C Bus Arbitration

Attach two different sensors (e.g. BME280 on 0x76 and an OLED SSD1306 display on 0x3C) to the same physical I2C1 bus. Verify that independent driver instances share the bus without collision.

Challenge 4 (STM32)

Circular Black-Box Flight Log in Flash

Allocate a 32KB circular logging partition in high STM32 flash memory. Record timestamped sensor fault telemetry and provide a UART dump command to read out crash logs after reboot.

Challenge 5 (CI/CD Automation)

Automated Code Coverage Badging in GitHub

Configure gcov and lcov in your GitHub Actions workflow to generate an HTML code coverage report, enforcing a mandatory 90% branch coverage threshold on all pull requests.
